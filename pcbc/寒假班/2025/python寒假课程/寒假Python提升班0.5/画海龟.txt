import turtle
turtle.speed(0.1)
def gk(a,r,j,w1,w2,w3):
    turtle.left(w1)
    turtle.forward(a)
    turtle.right(w2)
    turtle.circle(r, j)
    turtle.right(w3)
    turtle.forward(a)
turtle.right(90)
turtle.penup()
turtle.goto(-100, 100)
turtle.pendown()
turtle.circle(90, 360, 6)
gk(50,-140,60,270,90,90)
gk(50,-140,60,180,90,90)
gk(50,-140,60,180,90,90)
gk(50,-140,60,180,90,90)
gk(50,-140,60,180,90,90)
gk(50,-140,60,180,90,90)
turtle.penup()
turtle.goto(-150, 130)
turtle.pendown()
turtle.right(180)
turtle.circle(-50, 250)
turtle.done()
