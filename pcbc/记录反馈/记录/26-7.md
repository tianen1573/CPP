void selectionSort(){
	
	// 一共有n轮， i 代表 当前是第几轮， 排第几小 
	for(int i = 1; i <= n; i ++){
		int mini = i;
		
		for(int j = i; j <= n; j ++){
			if(arr[j] < arr[mini]){
				mini = j;
			}
		}
		
		swap(arr[i], arr[mini]);
		
	}

}

void insertSort(){
	// 一共n轮， i 表示轮次，也表示插入的位置 
	for(int i = 2; i <= n; i ++){
		int id = i - 1, key = arr[i];
		// 比arr[id] 小，顶掉他, 第id个往后移 
		while(id > 0 && key < arr[id]){
			arr[id + 1] = arr[id];
			id --; 
		} 
		arr[id + 1] = key;
	}
	
} 

void cntSort(){
	
	int minn = INT_MAX, maxx = INT_MIN;
	for(int i = 1; i <= n; i ++){
		cnt[arr[i]] ++;
		minn = min(minn, arr[i]);
		maxx = max(maxx, arr[i]);
	}
	
	for(int i = minn; i <= maxx; i ++){
		while(cnt[i] != 0){
			cout << i << ' ';
			cnt[i] --;
		}
	}
	
	
}

    ~~~

    	所以gcd(x, y) 和 gcd(y%x, x)等价

~~~C++
排列和组合的区别：
排列关系成分、顺序
组合只关心成分 

排列数：A(n, m)
从n个里面选m个的排列，有多少种方案

A(n, m) 
= n! / (n - m)!
= n * (n - 1) * ... * (n - m + 1)

从n开始乘上m个数

------

组合数：C(n, m)
从n个里面选m个的组合，有多少种方案

C(n, m) 
= A(n, m) / m! = n! / (m! * (n-m)!)
= [n * (n -1) * ... * (n - m + 1)] / [1 * 2 * .... * m]

公式推导，先假设是排列，则有A(n,m)方案，再去掉内部顺序m!
得到 A(n, m) / m!

分母：从n开始乘上m个数
分子：从1乘到m 

C(n, m) = C(n, n - m)
C(n, 0) = 1
杨辉三角的第i行第j列的值 = C(i, j), i,j 从0开始 
杨辉三角第i行元素的和 = 2^i

C(n, m) = C(n - 1, m - 1) + C(n - 1, m)
 
 
~~~

  # 推一下辗转相除法
  求 x , y 的 最大公约数， 假设：x <= y
  那么可得：y = k * x + a; k >= 1, a >= 0
  
  如果a是0：
  	说明 y就是x的倍数，那么此时最大公因子就是x

  	所以gcd(x, y) 和 gcd(y%x, x)等价
