~~~C++
#include<bits/stdc++.h>

	string s,t;
	cin >> s >> t;
	// s倒着存存到a数组，t倒着存存到b数组 
	for(int i = 0; i < s.size(); i++) a[s.size()-1-i]=s[i]-'0';
	for(int i = 0; i < t.size(); i++) b[t.size()-1-i]=t[i]-'0';
	int len = max(s.size(), t.size());
	for(int i = 0; i < len; i ++){
		c[i] += a[i]+b[i]; // 累加
		c[i+1] += c[i] / 10; // 进位
		c[i] %= 10; // 存余 
	}
	if(c[len] != 0) len++; // 可能会进位
	
	for(int i = len - 1; i >= 0; i --)
		cout << c[i];
~~~

# 高精度乘法

> 高精度*低精度模板

~~~C++
// 8258
#include<bits/stdc++.h>
using namespace std;
const int N = 1e5+11;
int a[N]; // a是存阶乘的
int main(){  
	int n;
	cin>>n;
	a[0]=1; // 初始化为1 
	int len=1; // 长度一开始为1 
	for(int k = 1; k <= n; k++){
		// 处理a，每一位都需要*k 
		for(int i=0;i<len;i++){ 
			a[i]*=k;
		}
		// 处理进位，不能同时进行 
		for(int i=0;i<len;i++){  
			a[i+1]+=a[i]/10;
			a[i]%=10;
			if(a[len] != 0) len++; // 可能会多一位 
		}
	}
	for(int i=len-1;i>=0;i--) cout<<a[i];
	return 0;
}
~~~

> 高精度乘法模板

~~~C++
#include<bits/stdc++.h>
using namespace std;
const int N = 1e5+11;
long long a[N], b[N], c[N]; // 定义成long long 
string s, t; 
int main(){
	cin>>s>>t; // 输入 
	int la=s.size(),lb=t.size(); // 记录长度 
	for(int i=0; i<la;i++) a[i]=s[la-1-i]-'0'; // s倒转存在a里 
	for(int i=0; i<lb;i++) b[i]=t[lb-1-i]-'0'; // t倒转存在b里 
	int lc=la+lb; // c的理论最大长度是 la+lb 
	for(int i=0;i<la;i++){ // 枚举a的位数 
		for(int j=0;j<lb;j++){ // 枚举b的位数 
			// a[i]*b[j] 对应c的位置是 c[i+j] 
			c[i+j]+=a[i]*b[j]; // 累加 
			c[i+j+1]+=c[i+j]/10; // 进位 
			c[i+j]%=10; // 取余 
		}
	}
	
	while(lc>0&&c[lc]==0) lc--; // 处理进位，直到遇见非0或者最后一个 
	// 3. 输出
	for(int i=lc;i>=0;i--){ // 此时lc是最高位的下标 
		cout<<c[i];
	}

	return 0;
}
~~~
