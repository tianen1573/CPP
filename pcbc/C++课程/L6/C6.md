// 从前k个草药里面选，总体积不超过vs的方案 的 价值最大和 
int dp[i][j]: 从前i个物品里面选，体积不超过j的 最大 价值和 
int dfs(int k, int vs){
	if(k < 1){
		return 0;
	}
	if(dp[k][vs] == -1){
		// 不选
		int maxx = INT_MIN;
		maxx = dfs(k - 1, vs);	
		// 选 
		if(vs >= v[k]){
			maxx = max(maxx, dfs(k - 1, vs - v[k]) + w[k])
		} 
		dp[k][vs] = maxx;
	} 
	return dp[k][vs];
}

dp[i][j] = max{dp[i - 1][j], dp[i - 1][j - v[i]] + w[i]}; i,j >= 1
