	// 1. 平均值 = 总和 / 数量

	// 3. 再次遍历
	int a[111];
	int sum = 0;
	for(int i = 1; i <= n; i ++){
		cin >> a[i];
		sum += a[i];
	}
	double avg = 1.0 * sum / n;
	for(int i = 1; i <= n; i ++){
		if(a[i] > avg){
			cout << a[i] << " ";

	}

		}

	// 没return -》》没找到

~~~C++
#include<bits/stdc++.h>
using namespace std;
int main(){
	
	// 1. 求n个元素的和
	int n;
	int arr[101];
	cin >> n;
	int sum = 0;
	for(int i = 1; i <= n; i ++){
		cin >> arr[i];
		
		sum += arr[i];
	}
	cout << sum;
    
    // 2. 求最大值
    /*
    	求最大值，需要定义一个变量用来存储
		并且，需要把最大值初始化成比数组的最小值还要小(或者初始化成第一个元素的大小) 
		然后，遍历数组，和每一个元素进行比较
		如果，a[i]大于maxx，说明a[i]是新的最大值 
	*/
	int arr[101];
	cin >> n;
	for(int i = 1; i <= n; i ++){
		cin >> arr[i];
	}
	int maxx = arr[1];
	for(int i = 1; i <= n; i ++){
		if(arr[i] > maxx){
			// 比我们当前找到的最大值还要大，他就是新的最大值
			maxx = arr[i]; 
		} 
	}
	cout << maxx;
    
    // 3. 求最小值
    /*
    	求最小值，需要定义一个变量用来存储
		并且，需要把最小值初始化成比数组的最大值还要大(或者初始化成第一个元素的大小) 
		然后，遍历数组，和每一个元素进行比较
		如果，a[i]小于minn，说明a[i]是新的最小值
	*/
	int arr[101];
	cin >> n;
	for(int i = 1; i <= n; i ++){
		cin >> arr[i];
	}
	int minn = arr[1];
	for(int i = 1; i <= n; i ++){
		if(arr[i] < minn){
			// 比我们当前找到的最小值还要小，他就是新的最小值
			minn = arr[i]; 
		} 
	}
	cout << minn;
    
    // 4. 求最大值最小值对应的下标
    /*
    	在最大值的基础上，又求了一个其对应的下标
		那我们就需要，再定义一个变量存其对应的下标 
		并且，在更新最大值的同时，更新对应的下标 
	*/
	int arr[101];
	cin >> n;
	for(int i = 1; i <= n; i ++){
		cin >> arr[i];
	}
	int maxx = arr[1]; // 初始化成第一个元素 
	int p = 1; // 那么， 此时最大值对应的下标应该是1 
	for(int i = 1; i <= n; i ++){
		if(arr[i] > maxx){
			// 比我们当前找到的最大值还要大，他就是新的最大值
			maxx = arr[i]; 
			p = i; // 还需要修改对应的坐标 
		} 
	}
	cout << maxx;
    
    // 5. 求去掉最大值最小值之后的分数（保证最大值最小值不同） 
    /*
    	可分解成三步：求和，求最大值，求最小值
	*/
	int arr[101];
	cin >> n;
	for(int i = 1; i <= n; i ++){
		cin >> arr[i];
	}
	int sum = 0; 
	int maxx = arr[1];
	int minn = arr[1]; 
	for(int i = 1; i <= n; i ++){
		if(arr[i] > maxx){
			maxx = arr[i];  // 求最大 
		} 
		if(arr[i] < minn){
			minn = arr[i]; // 求最小 
		}
		sum += arr[i]; //求和 
	}
	cout << sum - maxx - minn;
    
    // 6. 求次大值/次小值
	
} 
~~~

bool isP(int x){
	// 小于2的一定不是质数 
	if(x < 2) return 0;
	// 在 2 ~ sqrt(x)里面找有没有其他的因数
