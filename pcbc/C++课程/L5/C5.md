#### 排列组合

> 知识点：回溯、选数、方案和、放苹果

- 重点：选或者不选、回溯、剪枝

> 题目本质

输入n个元素，可以从其中选择k个(k<=n)，问：有多少种方案 ？

1. 如果和顺序有关：是排列；如果和顺序无关：是组合
2. 特殊要求：和最大的方案、和是素数、能否重复、能否不选等

> **知识点1**：排列、组合、全排列的概念

- 如果和顺序有关：是排列；
  ~~~C++
  1 2 3
  1 3 2
  是两种排列
  ~~~
- 如果和顺序无关：是组合；
  ~~~C++
  1 2 3
  1 3 2
  是同一种组合
  ~~~

- k == n ： 全排列

> 前提

算法是在模拟人的计算，先将简单的样例模拟出来，再找规则和统一的解决方法。

也就是说，纸上你都搞不定，代码肯定整不了。

~~~C++
知识点4：组合的顺序问题 

	问题来了：上面的代码会输出123，132这两中方案，可他们却是同一种组合。
	 
	组合和排列一样，是不能随意的选的，上述代码产生的123，132就是同一种组合。
	组合虽然不要求顺序，但求解的过程中，我们是需要给我们的方案指定一个顺序的。
	比如，我们要求，最终的方案是一个升序序列。
	那么，123、132、213、231、.....，就只有123是符合要求的方案 
	
	升序，就要求，我在进行选择时，本次选择的元素，要比上一个大。
	那么，我们就需要知道上一个是谁，也就需要多一个形参last，代表上一个位置的元素。 
	不再是，从1~n里面选，而是last+1~n里面选。 
	void dfs(int k, int last){
    	if(k > m){
            其他操作：输出、求和等 
            return ;
        }
		// 枚举所有可以的选择 
        for(int i = last + 1; i <= n; i ++){
            path[k] = i; // 选择的是i
            dfs(k + 1, i);
        }
	}

	只有一个参数k的时候，我们都知道dfs(1)从第一步启动。
	多了一个参数last的时候，要怎么启动呢，这个时候，要去考虑第一步能够选择什么。
	第一步没有限制，可以从1~n里面选。
	也就是说 for(int i = last + 1; i <= n; i ++) 里面的last + 1 == 1。
	也就是说，一开始传进来的last，应该是个0，即dfs(1, 0)
~~~

~~~C++
#include<bits/stdc++.h>
using namespace std;
int dx[]={0,1,0,-1},dy[]={1,0,-1,0}; // 四个方向 
struct node{
	int x,y;
};
queue<struct node> que;
int mmp[45][45];
int st[45][45];
int n;
bool check(int x,int y)
{
	// 现实中给了我们一个迷宫，里面有一些封闭的区域，那怎么找到这个区域呢？
	// 我们可以找一个入口，往里面灌水，水不能达到的地方就是封闭的。
	// 那这一道题，怎么找到这个入口呢？
	// 可以发现，枚举二维数组的外层开始灌水，就可以找到所有封闭的区域。
	// 但这样操作，我们需要判断所有的最外层，要入队的比较多。
	// 其实，可以从（0，0）开始灌水，是可以通过这个点来蔓延至所有的最外层的。
	// 那就会导致，合法区域往外面扩了一圈。 
	if(x<0||x>n+1||y<0||y>n+1) return 0; // 越界 
	if(mmp[x][y]==1) return 0; // 墙 
	if(st[x][y]==1) return 0; // 走过 
	return 1;
 } 
int main(){
	cin>>n;
	for(int i=1;i<=n;i++){
		for(int j=1;j<=n;j++){
			cin>>mmp[i][j];
		}
	 } 
	 // 入队(0,0) 
	 que.push({0,0});
	 st[0][0]=1;
	 // 开始广搜 
	 while(que.size()){
	 	struct node nd=que.front();
	 	que.pop();
	 	for(int i=0;i<4;i++)
	 	{
	 		int nx=nd.x+dx[i];
	 		int ny=nd.y+dy[i];
	 		if(check(nx,ny)==true){
				 que.push({nx,ny});
				 st[nx][ny]=1;
			 }
		 }
	 }
	 
	 for(int i=1;i<=n;i++){
		for(int j=1;j<=n;j++){
			// 原来的是1，现在只能为1 
			if(mmp[i][j] == 1) cout << 1 << " ";
			else{
				// 原来的是0，但能被水经过，所有不是封闭的 
				if(st[i][j] == 1) cout << 0 << " ";
				// 原来的是0，并且不能被水经过，所有是封闭的
				else cout << 2 << " ";
			}
		}
		cout << endl;
	}
} 
~~~

### 连通块大小

> 

![image-20241130222003468](%E5%9B%BE%E7%89%87/C4/image-20241130222003468.png)

~~~C++
#include<bits/stdc++.h>
using namespace std;
int dx[]={0,1,0,-1},dy[]={1,0,-1,0}; // 四个方向 
struct node{
	int x,y;
};
queue<struct node> que;
char mmp[145][145];
int st[145][145];
int n,m;
bool check(int x,int y)
{
	if(x<1||x>n||y<1||y>m) return 0; // 越界 
	if(st[x][y]==1) return 0; // 走过 
	return 1;
 } 
int main(){
	cin>>n>>m;
	for(int i=1;i<=n;i++){
		for(int j=1;j<=m;j++){
			cin>>mmp[i][j];
		}
	 } 
	 int ans = 0;
	 for(int i = 1; i <= n; i ++){
	 	for(int j = 1; j <= m; j ++){
	 		if(check(i, j) == 1){ // 没有被走过
			 	// 入队，开走 
	 			st[i][j] = 1;
	 			que.push({i, j});
	 			// 需要统计当前连通块的大小 
	 			int cnt = 0;
	 			// 根据题意可得，当前深搜能够达到的点一定是同一个连通块的
				// 那么while循环执行了多少次，就代表有多少个节点入队了 
	 			while(que.size() > 0){
	 				cnt ++;
	 				struct node nd = que.front();
	 				que.pop();
	 				for(int i = 0; i < 4; i ++){
	 					int nx = nd.x + dx[i];
	 					int ny = nd.y + dy[i];
	 					// 不仅要没走过，还要值相同 
	 					if(check(nx, ny) && mmp[nx][ny] == mmp[nd.x][nd.y]){
	 						st[nx][ny] = 1;
	 						que.push({nx, ny});
						}
					 }
				 }
				 // 更新最大值 
				 ans = max(ans, cnt);
			 }
		 }
	 }
	cout << ans;
	 
	 
} 
~~~

~~~C++
#include<bits/stdc++.h>
#define ll long long
using namespace std;

int dx[] = {-1, 0 , 1, 0}, dy[] = {0, 1, 0, -1};
char mmp[1001][1001];
int st[1001][1001];
int cnt[1000003];
int n, m, k, cx, cy;
struct node{
	int x, y;
};
queue<node> que;

bool check(int x, int y){
	if(x < 1 || x > n || y < 1 || y > n) return 0;
	if(st[x][y] != 0) return 0;
	return 1;
}

void bfs(int sx, int sy){
	st[sx][sy] = k;
	que.push({sx, sy});
	while(que.size()){
		cnt[k] ++;
		node nd = que.front();
		que.pop();
		for(int i = 0; i < 4; i ++){
			int nx = nd.x + dx[i];
			int ny = nd.y + dy[i];
			if(check(nx, ny) && mmp[nd.x][nd.y] != mmp[nx][ny]){
				st[nx][ny] = k;
				que.push({nx, ny});
			}
		}
	}
}
int main(){
	
	cin >> n >> m;
	for(int i = 1; i <= n; i ++){
		for(int j = 1; j <= n; j ++){
			cin >> mmp[i][j];
		}
	}
	for(int i = 1; i <= n; i ++){
		for(int j = 1; j <= n; j ++){
			if(st[i][j] == 0){
				++ k;
				bfs(i, j);
			}
		}
	}
	for(int i = 1; i <= m; i ++){
		cin >> cx >> cy;
		cout << cnt[st[cx][cy]] << endl;
	}
	
	
}
~~~
